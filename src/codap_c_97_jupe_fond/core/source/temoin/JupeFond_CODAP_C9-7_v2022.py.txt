def read_setting_file(filename):
    with open(filename, 'r') as f:
        lines = f.readlines()
        setting_data = list()
        endfile = False
        i = 0
        while not endfile:
            line = lines[i]
            # print(line)
            if line.startswith("$$Start"):
                j = 1
                endkeyword = False
                while not endkeyword:
                    content=lines[i + j+1]
                    if content.startswith("$$End"):
                        i = i + j
                        endkeyword = True					
                    else:
                        if content.startswith("\n"):						
                            j = j + 1	
                        else:
                            setting_data.append(content.strip('\n'))
                            j = j + 1										
            i = i +1
            if i == len(lines):
                endfile = True
	
    return setting_data

def get_data_set(items,arg):

    param = {}

    for item in items:
        for i in range(len(arg)):
            if "$"+item in arg[i]:
                param[item]=arg[i+1]

    return param

def loding_parameters():
    '''
    Etape 3 : Calcul des valeurs des parametres qui décrivent les sollicitations appliquées au raccordement (la jonction)

    '''
    
    global R1, R2, R3, R4, R5, R6, R7
    
    R1 = p*D3/(2*eu1)
    
    R2 = 2*FZ/(pi*D3*eu1)
    
    R3 = 4*MZ/(pi*eu1*D3**2)
    
    R4 = c/eu1*(0.5*R2 + R3)
    
    if situation == 'autre':
        R5 = (deltaTK*alphaK - deltaTT*alphaT)*E
        R6 = (alphaT*gT + alphaK*gK)*E*np.sqrt(0.5*D3*eu1)
        R7 = (gK + gB)*alphaK*E*np.sqrt(0.5*D3*eu1)
    elif situation == 'essai':
        R5 = np.zeros(E.shape)
        R6 = np.zeros(E.shape)
        R7 = np.zeros(E.shape)

def get_raccord_C_fond_elliptique_C21():   
    '''
    
    Etape 4 : Raccordement type C et fond elliptique : Calcul des valeurs des coefficients Cij

    '''
    global C21
    
    C21 = list()
    
    for i in range(len(lambda1)):
    
        x = lambda1[i]
        
        if delta1[i] <=0.6 :
            z1 = 0.6
            y1 = 14.9631 - 8.3866*log(x) + 1.4621*log(x)**2 - 0.0902*log(x)**3
            z2 = 1.2
            y2 = 8.3022 - 4.3673*log(x) + 0.7038*log(x)**2 - 0.0414*log(x)**3
        
        elif 0.6 < delta1[i] and delta1[i] <=0.8:
            z1 = 0.6
            y1 = 14.9631 - 8.3866*log(x) + 1.4621*log(x)**2 - 0.0902*log(x)**3
            z2 = 0.8
            y2 = 10.7904 - 5.7860*log(x) + 0.9459*log(x)**2 - 0.0562*log(x)**3
            
        elif 0.8 < delta1[i] and delta1[i] <=1.0:
            z1 = 0.8
            y1 = 10.7904 - 5.7860*log(x) + 0.9459*log(x)**2 - 0.0562*log(x)**3
            z2 = 1.0
            y2 = 8.2973 - 4.2878*log(x) + 0.6642*log(x)**2 - 0.0382*log(x)**3
        
        elif 1.0 < delta1[i] and delta1[i] <=1.2:
            z1 = 1.0
            y1 = 8.2973 - 4.2878*log(x) + 0.6642*log(x)**2 - 0.0382*log(x)**3
            z2 = 1.2
            y2 = 8.3022 - 4.3673*log(x) + 0.7038*log(x)**2 - 0.0414*log(x)**3
    
        elif 1.2 < delta1[i] :
            z1 = 0.6
            y1 = 14.9631 - 8.3866*log(x) + 1.4621*log(x)**2 - 0.0902*log(x)**3
            z2 = 1.2
            y2 = 8.3022 - 4.3673*log(x) + 0.7038*log(x)**2 - 0.0414*log(x)**3
            
        a = (y1 - y2)/(z1 - z2)
        b = y1 - a*z1
        
        C21.append(a*delta1[i] + b)
        
        
    C21 = np.array(C21).reshape(delta1.shape)
